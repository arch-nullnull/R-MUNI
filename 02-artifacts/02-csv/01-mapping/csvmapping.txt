# ==========================================================
# csvmapping.txt
# Declarative Scope Mapping for CSV Flow
#
# Purpose:
# - Define WHICH source models are included in the current run
# - Acts as HARD FILTER for CSV03-resolve_run_scope
#
# Principles:
# ----------------------------------------------------------
# - Mapping is configuration, not logic
# - Rules are OR-combined
# - If NO rule matches -> model is NOT in scope
# - CSV03 extracts ONLY the model selector (model=...)
# - Element/type selectors are ignored by CSV03
#
# Supported dimensions (current):
# - source system (archi, masterxml, xlsx, ...)
# - model name pattern (wildcards supported)
#
# Syntax:
# <source>[model=<pattern>]-<ignored-part>
#
# Notes:
# - <pattern> supports wildcards (*)
# - Matching is substring-based
# - CSV03 does NOT interpret element/type filters
# ==========================================================


# ----------------------------------------------------------
# ACTIVE RULES (TEST SETUP)
# ----------------------------------------------------------
# Include ALL models from ALL sources
# (useful for initial wiring tests)
#
archi[model=*.*]-element-xsi:type="*"

# REAL-WORLD EXAMPLES (INACTIVE / DOCUMENTATION)
