# ==========================================================
# M2Bmapping.txt
# Filter rules for Master → BPMN (M2B)
#
# Purpose:
# - Define WHICH Archi BusinessProcess elements
#   are materialized as BPMN processes
#
# Scope:
# - Applied ONLY in M2B01 (master extract)
# - Acts as HARD FILTER
#
# Semantics:
# - If NO rule matches → element is ignored
# - Rules act ONLY as selectors (no transformation)
# - If a rule matches → COMPLETE element is processed
# - No IDs are used as filter criteria
#
# Supported filter dimensions:
# - Archi source model name (sourceModel)
# - Element xsi:type
#
# Syntax:
# archi[model=<pattern>]-element-xsi:type="<Type>"
#
# Notes:
# - <pattern> supports wildcards (*)
# - Matching is substring-based
# - Multiple rules are OR-combined
# ==========================================================

archi[model="MUNI FLOW.xml"]-element-xsi:type="BusinessProcess"
